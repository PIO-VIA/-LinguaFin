/**
 * Formate un identifiant de jeton (base 1) selon le format configuré.
 */
export function formatToken(index1Based: number, format: TokenFormat = DEFAULT_TOKEN_FORMAT): string {
  if (format === "xml-tag") {
    return `<x id="${index1Based}"/>`;
  }
  return `{{${index1Based}}}`;
}
 
/**
 * Vérifie si deux intervalles [start, end[ se chevauchent.
 */
function overlaps(a: MatchSpan, b: MatchSpan): boolean {
  return a.start < b.end && b.start < a.end;
}

/**
 * Restaure les jetons "{{n}}" ou "<x id=\"n\"/>" par leurs valeurs d'origine (par numéro de jeton).
 * Effectue le remplacement en une seule passe afin qu'aucune valeur restaurée ne soit ré-interprétée.
 */
export function restore(translated: string, tokens: string[]): string {
  if (!translated || tokens.length === 0) {
    return translated;
  }
 
  return translated.replace(/\{\{(\d+)\}\}|<x\s+id="(\d+)"\s*\/>/g, (fullMatch, braceIdx, xmlIdx) => {
    const rawIndex = braceIdx ?? xmlIdx;
    const index1Based = Number.parseInt(rawIndex, 10);
    if (Number.isNaN(index1Based) || index1Based < 1 || index1Based > tokens.length) {
      return fullMatch;
    }
    return tokens[index1Based - 1];
  });
